[ -z "$DEFAULT_IP" ] && DEFAULT_IP="192.168.1.1"

LAN_IP="$USER_IP"

# Stage the network interface configuration change via UCI (Do NOT restart network yet)
echo "[*] Staging network interface lan update to: ${LAN_IP}..."
if [ -f /etc/config/network ]; then
    uci set network.lan.proto='static'
    uci set network.lan.ipaddr="$LAN_IP"
    uci commit network
fi

# 1. Prepare web space
mkdir -p /www/cgi-bin

# Set default master password for hotel owner if not configured
[ ! -f /etc/portal_secret ] && echo "admin123" > /etc/portal_secret

# 2. Configure Web Server to Catch and Redirect Canary Requests (With LuCI Hijack Fix)
echo "[*] Optimizing web server for captive portal detection..."
if [ -f /etc/config/uhttpd ]; then
    uci del uhttpd.main.lua_prefix 2>/dev/null
    uci del uhttpd.main.lua_handler 2>/dev/null
    uci del uhttpd.main.ubus_prefix 2>/dev/null
    
    uci set uhttpd.main.index_page='cgi-bin/index'
    uci set uhttpd.main.error_page='/cgi-bin/index'
    uci commit uhttpd
    /etc/init.d/uhttpd restart 2>/dev/null
fi

# 3. Handle DNS/DHCP Settings & Inject Dynamic Boot script
echo "[*] Optimizing DNS/DHCP settings..."
if [ -f /etc/config/dhcp ]; then
    # Clean up old Option 114 formats if they exist
    uci del_list dhcp.lan.dhcp_option="114,http://192.168.1.1/cgi-bin/index" 2>/dev/null
    uci del_list dhcp.lan.dhcp_option="114,http://${DEFAULT_IP}/cgi-bin/index" 2>/dev/null
    uci del_list dhcp.lan.dhcp_option="114,http://${LAN_IP}/cgi-bin/index" 2>/dev/null
    
    # Inject the immediate target option for this current session run
    uci add_list dhcp.lan.dhcp_option="114,http://${LAN_IP}/cgi-bin/index"
    uci set dhcp.@dnsmasq[0].rebind_protection='0'
    uci commit dhcp
fi

echo "[*] Creating dynamic boot updater for DHCP option 114..."
cat << 'BOOT_EOF' > /etc/init.d/portal_dhcp
#!/bin/sh /etc/rc.common

START=19

boot() {
    # Dynamically find what LAN IP the router actually picked up on boot
    LIVE_IP=$(ubus call network.interface.lan status 2>/dev/null | grep -A2 '"ipv4-address"' | grep '"address"' | awk -F'"' '{print $4}')
    [ -z "$LIVE_IP" ] && LIVE_IP=$(ifconfig br-lan 2>/dev/null | grep 'inet addr:' | awk -F: '{print $2}' | awk '{print $1}')
    [ -z "$LIVE_IP" ] && LIVE_IP="192.168.1.1"

    # Delete any legacy or mismatch entries safely from memory array
    while uci show dhcp.lan.dhcp_option 2>/dev/null | grep -q "114,"; do
        OLD_OPT=$(uci show dhcp.lan.dhcp_option 2>/dev/null | grep "114," | head -n 1 | awk -F"='" '{print $2}' | sed "s/'//g")
        [ -n "$OLD_OPT" ] && uci del_list dhcp.lan.dhcp_option="$OLD_OPT"
    done

    # Add the current true IP translation route and commit
    uci add_list dhcp.lan.dhcp_option="114,http://${LIVE_IP}/cgi-bin/index"
    uci commit dhcp
    /etc/init.d/dnsmasq restart >/dev/null 2>&1
}
BOOT_EOF

    uci set firewall.@rule[-1].src='lan'

    uci set firewall.@rule[-1].src='lan'
    uci set firewall.@rule[-1].dest_port='443'
    uci set firewall.@rule[-1].proto='tcp'
    uci set firewall.@rule[-1].target='REJECT'

    uci add firewall redirect
    uci set firewall.@redirect[-1].name='portal_http_intercept'
    uci set firewall.@redirect[-1].src='lan'
    uci set firewall.@redirect[-1].src_dport='80'
    uci set firewall.@redirect[-1].proto='tcp'
    uci set firewall.@redirect[-1].target='DNAT'
    uci set firewall.@redirect[-1].dest='lan'

    # the router itself, regardless of whatever future IP changes happen on LAN.
    uci set firewall.@redirect[-1].dest_port='80'

            if [ "$PASS_INPUT" = "$ADMIN_PASS" ]; then
                CURRENT_TOKEN=$(dd if=/dev/urandom bs=1 count=16 2>/dev/null | hexdump -e '"%02x"')
                echo "$CURRENT_TOKEN" > $TOKEN_FILE
                echo "Set-Cookie: session_id=$CURRENT_TOKEN; Path=/; HttpOnly"
                JUST_LOGGED_IN="true"
            fi

            TARGET_MAC=$(echo "$POST_DATA" | sed -n 's/.*target_mac=\([^&]*\).*/\1/p' | sed 's/%3A/:/g' | tr 'A-Z' 'a-z')
            if [ -n "$TARGET_MAC" ]; then
                iptables -t nat -D prerouting_lan_rule -m mac --mac-source "$TARGET_MAC" -j ACCEPT 2>/dev/null
                iptables -D forwarding_lan_rule -m mac --mac-source "$TARGET_MAC" -j ACCEPT 2>/dev/null
                ip6tables -D forwarding_lan_rule -m mac --mac-source "$TARGET_MAC" -j ACCEPT 2>/dev/null

                    sed -i "/$TARGET_MAC/d" /etc/firewall.user
